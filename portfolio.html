#include <iostream>
#include <fstream>
#include <iomanip>
using namespace std;

// ATM Machine Simulator by Shubham Rajpoot
class ATM {
private:
    int pin = 1234;
    double balance = 25000.00;
    int attempts = 0;

public:
    bool verifyPin(int enteredPin) {
        if (enteredPin == pin) {
            attempts = 0;
            return true;
        }
        attempts++;
        return false;
    }

    void checkBalance() {
        cout << "Current Ledger Balance: ₹ " << fixed << setprecision(2) << balance << endl;
    }

    bool withdraw(double amount) {
        if (amount <= 0 || amount > balance) return false;
        balance -= amount;
        logTransaction("WITHDRAWAL", amount);
        return true;
    }

    bool deposit(double amount) {
        if (amount <= 0) return false;
        balance += amount;
        logTransaction("DEPOSIT", amount);
        return true;
    }

    void logTransaction(string type, double amount) {
        ofstream ledger("atm_history.txt", ios::app);
        if (ledger.is_open()) {
            ledger << type << ": ₹ " << amount << " | Balance: ₹ " << balance << endl;
            ledger.close();
        }
    }
};
